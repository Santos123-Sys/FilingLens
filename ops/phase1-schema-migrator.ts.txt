import mysql from "mysql2/promise";
let status = "starting";
let detail = "migration not yet attempted";
async function migrate() {
 const url = Bun.env.DATABASE_URL;
 if (!url) throw new Error("DATABASE_URL missing");
 const db = await mysql.createConnection(url);
 try {
   const statements = [
`CREATE TABLE IF NOT EXISTS companies (
 id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
 jurisdiction VARCHAR(2) NOT NULL,
 registry_id VARCHAR(64) NOT NULL,
 legal_name VARCHAR(255) NOT NULL,
 ticker VARCHAR(32) NULL,
 created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
 UNIQUE KEY uq_company_identity (jurisdiction, registry_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4`,
`CREATE TABLE IF NOT EXISTS filings (
 id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
 company_id BIGINT UNSIGNED NOT NULL,
 filing_key VARCHAR(191) NOT NULL,
 form_type VARCHAR(64) NOT NULL,
 period_end VARCHAR(10) NOT NULL,
 filed_at VARCHAR(10) NOT NULL,
 source_url TEXT NULL,
 created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
 UNIQUE KEY uq_company_filing (company_id, filing_key),
 KEY ix_filing_period (company_id, period_end),
 CONSTRAINT fk_filings_company FOREIGN KEY (company_id) REFERENCES companies (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4`,
`CREATE TABLE IF NOT EXISTS financial_observations (
 id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
 filing_id BIGINT UNSIGNED NOT NULL,
 metric VARCHAR(100) NOT NULL,
 period_end VARCHAR(10) NOT NULL,
 period_kind VARCHAR(3) NOT NULL,
 fiscal_year INT NOT NULL,
 fiscal_quarter INT NULL,
 value VARCHAR(128) NULL,
 currency VARCHAR(8) NULL,
 unit VARCHAR(64) NOT NULL,
 source_section TEXT NOT NULL,
 source_url TEXT NULL,
 created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
 KEY ix_observation_filing (filing_id),
 KEY ix_observation_period (metric, period_kind, fiscal_year),
 CONSTRAINT fk_observations_filing FOREIGN KEY (filing_id) REFERENCES filings (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4`
,`CREATE TABLE IF NOT EXISTS regulatory_snapshots (
 id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
 company_id BIGINT UNSIGNED NOT NULL,
 provider VARCHAR(64) NOT NULL,
 snapshot_day VARCHAR(10) NOT NULL,
 snapshot_hash VARCHAR(64) NOT NULL,
 payload_json TEXT NOT NULL,
 created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
 UNIQUE KEY uq_reg_snapshot_day (company_id, snapshot_day),
 KEY ix_reg_snapshot_company (company_id, created_at),
 CONSTRAINT fk_reg_snapshots_company FOREIGN KEY (company_id) REFERENCES companies (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4`
,`CREATE TABLE IF NOT EXISTS private_analysis_history (
 id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
 workspace_hash VARCHAR(64) NOT NULL,
 filing_key_hash VARCHAR(64) NOT NULL,
 jurisdiction VARCHAR(2) NOT NULL,
 issuer_name VARCHAR(255) NOT NULL,
 period_end VARCHAR(32) NOT NULL,
 payload_json TEXT NOT NULL,
 saved_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
 UNIQUE KEY uq_private_history (workspace_hash, filing_key_hash),
 KEY ix_private_workspace (workspace_hash, saved_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4`
,`CREATE TABLE IF NOT EXISTS sec_companyfacts_snapshots (
  cik CHAR(10) NOT NULL,
  retrieved_day CHAR(10) NOT NULL,
  archive_sha256 CHAR(64) NOT NULL,
  payload_sha256 CHAR(64) NOT NULL,
  source_url VARCHAR(255) NOT NULL,
  payload_json LONGTEXT NOT NULL,
  imported_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY(cik),
  KEY ix_sec_import_age(retrieved_day)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4`
   ];
   for(const sql of statements) await db.query(sql);
   const [tables] = await db.query("SELECT table_name FROM information_schema.tables WHERE table_schema=DATABASE() AND table_name IN ('companies','filings','financial_observations','regulatory_snapshots','private_analysis_history') ORDER BY table_name");
   const [keys] = await db.query("SELECT table_name, referenced_table_name FROM information_schema.key_column_usage WHERE table_schema=DATABASE() AND referenced_table_name IS NOT NULL AND table_name IN ('filings','financial_observations','regulatory_snapshots')");
   if ((tables as any[]).length!==5 || (keys as any[]).length!==3) throw new Error("Schema verification failed");
   const password=Bun.env.APP_DB_PASSWORD;
   if(!password || password.length<32) throw new Error("APP_DB_PASSWORD is required");
   await db.query("CREATE USER IF NOT EXISTS 'filinglens_app'@'%' IDENTIFIED BY "+mysql.escape(password));
   await db.query("ALTER USER 'filinglens_app'@'%' IDENTIFIED BY "+mysql.escape(password));
   await db.query("GRANT SELECT, INSERT, UPDATE, DELETE ON \`"+(new URL(url).pathname.slice(1).replace(/\`/g,""))+"\`.* TO 'filinglens_app'@'%'");
   const [grants]=await db.query("SHOW GRANTS FOR 'filinglens_app'@'%'");
   console.log("PHASE1_RESTRICTED_USER_VERIFIED",Array.isArray(grants) && grants.length>0);


   const backupPassword=Bun.env.BACKUP_DB_PASSWORD;
   if(!backupPassword || backupPassword.length<32) throw new Error("BACKUP_DB_PASSWORD missing");
   await db.query("CREATE USER IF NOT EXISTS 'filinglens_backup'@'%' IDENTIFIED BY "+mysql.escape(backupPassword));
   await db.query("ALTER USER 'filinglens_backup'@'%' IDENTIFIED BY "+mysql.escape(backupPassword));
   const database=new URL(url).pathname.slice(1);
   if(!/^[A-Za-z0-9_]+$/.test(database))throw new Error("Unsafe database identifier");
   await db.query("GRANT SELECT ON "+database+".* TO 'filinglens_backup'@'%'");
   const [backupGrants]=await db.query("SHOW GRANTS FOR 'filinglens_backup'@'%'");
   console.log("PHASE1_BACKUP_READER_VERIFIED",Array.isArray(backupGrants)&&backupGrants.length>0);
   status="ready";
   detail="5 tables and 3 foreign keys verified";
   console.log("PHASE1_SCHEMA_VERIFIED",JSON.stringify({tables:(tables as any[]).map(t=>t.TABLE_NAME??t.table_name),foreignKeys:(keys as any[]).length}));
 } finally {await db.end();}
}
try { await migrate(); } catch(e) {status="failed";detail=String(e);console.error("PHASE1_SCHEMA_FAILED",detail);}
Bun.serve({port:Number(Bun.env.PORT??3000),fetch(req){return new Response(JSON.stringify({status,detail}),{status:status==="ready"?200:503,headers:{"content-type":"application/json"}})}});
